\section{SSMQ Architecture, Algorithms, and Formal Analysis}
\label{sec:methodology}

Figure~\ref{fig:architecture} overviews \textbf{SSMQ}. Warm-up commits populate a historical co-change graph $\hat{G}$ and P90 risk scales. Pending retries, exact-tree stale-pass rechecks, blame splits, and bisection sub-batches preempt fresh batches across $W$ shared workers. Passing runs must match the current target branch ($M=M_0$) before commit; failing runs traverse a two-regime flaky gate and, once retries are denied or exhausted on multi-PR batches, invoke 1-step AST/traceback ranking or bisection fallback.

\subsection{Leak-Free Warm-Up Dependency Graph and Risk Scoring}
SSMQ constructs $\hat{G}=(V,\hat{E})$ strictly from the chronological warm-up prefix (earliest $25\%$ of merged PRs per repository, ordered by merge time and PR ID). For PR $p$ touching files $F(p)$, let $\mu(f)$ denote the two-level directory prefix of $f$, or the shared empty tuple $()$ for root files. From warm-up commits, SSMQ indexes all co-change sets $\mathcal{N}_{\text{cochange}}(f)$ (with no file-count filter) and directory neighborhoods $\mathcal{N}_{\text{mod}}(\mu(f))$, defining the 1-hop observed \emph{impact cone}:
\begin{equation}
C(p) = F(p) \cup \bigcup_{f \in F(p)} \Big(\mathcal{N}_{\text{cochange}}(f) \cup \mathcal{N}_{\text{mod}}(\mu(f))\Big).
\label{eq:cone}
\end{equation}

SSMQ computes a bounded risk score $r(p) \in [0, 1]$ from three log-transformed features normalized by floored nearest-rank warm-up P90 scales ($s = \max(1, \text{P90}_{\text{warm}})$) and clipped at $1$:
\begin{equation}
r(p) = 0.45\,\tilde{\ell}(p) + 0.30\,\tilde{f}(p) + 0.25\,\tilde{c}(p),
\label{eq:risk}
\end{equation}
where $\tilde{\ell}(p)=\min(1,\log(1+\text{add}+\text{del})/s_\ell)$, $\tilde{f}(p)=\min(1,\log(1+\max(\text{changed\_files},|F(p)|))/s_f)$, and $\tilde{c}(p)=\min(1,\log(1+\max_{f\in F(p)} c_{\text{warm}}(f))/s_c)$, with $c_{\text{warm}}(f)=|C(\{f\})|-1$ for warm-up-indexed paths and $0$ for unseen paths ($s_c$ is the P90 over warm-up-indexed paths). Defect priors calibrate strictly against warm-up mean risk $\bar{r}_{\text{warm}}$ (default $0.5$ on empty history, floored at $10^{-6}$) using a fixed generic prior rate $\hat{q}_0=0.10$ decoupled from the environment's true defect rate: $\hat{q}_i = \min(1, \max(10^{-6}, r(p_i)) \cdot \hat{q}_0 / \max(10^{-6}, \bar{r}_{\text{warm}}))$.

\subsection{Graph-Disjoint Risk-Stratified Multi-Lane Packing}
When a worker is idle and no pending job awaits dispatch, Algorithm~\ref{alg:packing} forms a fresh batch anchored at ready-queue head $p_h = Q[0]$, updating $Q$ and promoted set $\mathcal{P}$. If $p_h \in \mathcal{P}$ or $r(p_h) \ge \theta_{\text{risk}} = 0.55$, SSMQ opens a \textbf{Protected Lane} batch capped at $L=\min(2, K_{\max})$; otherwise it opens a \textbf{Fast Lane} batch capped at $L=K_{\max}$ (nominally $8$). Scanning $Q[1..]$ in FIFO order, Algorithm~\ref{alg:packing} promotes any PR overlapping accumulated cone $U$ into $\mathcal{P}$ (even when full or opposite-lane) and admits $p$ iff $|B|<L$, $U\cap C(p)=\emptyset$, and $p$'s lane matches $p_h$'s. Both lanes enforce cone disjointness within a batch. Simultaneous batches on different workers may overlap; scan cost scales with ready-queue length and cone operations (unmeasured).

\begin{algorithm}[t]
\caption{Graph-Disjoint Risk-Stratified Batch Packing}
\label{alg:packing}
\begin{algorithmic}[1]
\REQUIRE Queue $Q$, cones $C(\cdot)$, risk $r(\cdot)$, promoted $\mathcal{P}$, cap $K_{\max}$, $\theta_{\text{risk}}=0.55$
\ENSURE Disjoint batch $B$, updated queue $Q$, updated $\mathcal{P}$
\STATE $p_h \leftarrow Q[0]$ \COMMENT{Head anchor bounds fresh dispatches}
\STATE $\text{prot} \leftarrow (p_h \in \mathcal{P}) \lor (r(p_h) \ge \theta_{\text{risk}})$
\STATE $L \leftarrow \min(2, K_{\max})$ \textbf{if} $\text{prot}$ \textbf{else} $K_{\max}$
\STATE $B \leftarrow [p_h]$; \quad $U \leftarrow C(p_h)$; \quad $Q' \leftarrow []$
\FORALL{$p \in Q[1..]$}
    \STATE $\text{overlap} \leftarrow (U \cap C(p) \neq \emptyset)$
    \IF{$\text{overlap}$}
        \STATE $\mathcal{P} \leftarrow \mathcal{P} \cup \{p\}$ \COMMENT{Promote overlapping PR}
    \ENDIF
    \STATE $p_{\text{prot}} \leftarrow (p \in \mathcal{P}) \lor (r(p) \ge \theta_{\text{risk}})$
    \IF{$|B| < L \land \neg \text{overlap} \land (p_{\text{prot}} = \text{prot})$}
        \STATE $B.\text{append}(p)$; \quad $U \leftarrow U \cup C(p)$
    \ELSE
        \STATE $Q'.\text{append}(p)$
    \ENDIF
\ENDFOR
\STATE $Q \leftarrow Q'$; \quad $\mathcal{P} \leftarrow \mathcal{P} \setminus B$
\RETURN $B$
\end{algorithmic}
\end{algorithm}

\subsection{Exact-Tree Commit Check and Two-Regime Beta-Binomial Gate}
SSMQ maintains a plug-in conjugate $\mathrm{Beta}(\alpha,\beta)$ heuristic initialized at $(\alpha_0=1, \beta_0=19)$ (20 pseudocounts, prior mean $5\%$). Confirmed non-stale passes increment $\beta$ by $1$ and add that job's prior failures to $\alpha$; split, stale, and rejected runs are censored (a limitation discussed in Section~\ref{sec:threats}). On failure $m \in \{1, 2\}$ (at most two retries, i.e., three attempts), the Beta-Binomial predictive probability of $m$ consecutive flakes is:
\begin{equation}
P_{\text{flake}}(m \mid \alpha, \beta) = \prod_{k=0}^{m-1} \frac{\alpha + k}{\alpha + \beta + k}.
\label{eq:betabinom}
\end{equation}
Combining $P_{\text{flake}}$ with on-graph batch risk $\hat{q}(B)=1-\prod_{i\in B}(1-\hat{q}_i)\prod_{(i,j)\in\hat{E}(B)}(1-\hat{p}_c)$ using fixed generic conflict prior $\hat{p}_c=0.10$ (never reading environment fault rates) yields the positive-guarded transient score:
\begin{equation}
P(\text{transient} \mid \text{fail}_m) = \frac{(1 - \hat{q}(B))\, P_{\text{flake}}(m \mid \alpha, \beta)}{\hat{q}(B) + (1 - \hat{q}(B))\, P_{\text{flake}}(m \mid \alpha, \beta)}.
\label{eq:posterior}
\end{equation}
Algorithm~\ref{alg:flaky} applies asymmetric cutoffs: \textbf{Singleton Guard} ($\tau_1=0.25$, $|B|=1$) retries low-risk singletons before rejection, whereas \textbf{Batch Guard} ($\tau_B=0.50$, $|B|>1$) suppresses full-batch reruns because splitting preserves clean PRs. Before committing any passing job $J$ tested against snapshot $M_0$, Algorithm~\ref{alg:flaky} enforces \textbf{exact-tree commit authority}: it verifies $M = M_0$ (equivalently, that the verified tree $M_0 \cup J.B$ equals the target tree $M \cup J.B$). No commit-time conflict oracle is consulted. If another worker advanced mainline ($M \neq M_0$), the stale pass is invalidated and prepended to pending for fresh verification atop $M$, keeping retry count and round.

\begin{algorithm}[t]
\caption{Exact-Tree Commit Check \& Two-Regime Flaky Gate}
\label{alg:flaky}
\begin{algorithmic}[1]
\REQUIRE Completed run $R = (J, \text{pass}, M_0, T)$, mainline $M$, state $(\alpha, \beta)$
\IF{$R.\text{pass}$}
    \IF{$M \neq M_0$}
        \STATE Invalidate stale pass; prepend $J$ to pending (keeping failures/round)
        \RETURN
    \ENDIF
    \STATE $\alpha \leftarrow \alpha + J.\text{failures}$; \quad $\beta \leftarrow \beta + 1$; \quad $M \leftarrow M \cup J.B$
\ELSE
    \STATE $\tau \leftarrow 0.25$ \textbf{if} $|J.B| = 1$ \textbf{else} $0.50$
    \IF{$J.\text{failures} < 2 \land P(\text{transient} \mid \text{fail}_{J.\text{failures}+1}) \ge \tau$}
        \STATE Prepend retry ($J.\text{failures}{+}{+}$, $J.\text{round}{+}{+}$, reset overhead)
    \ELSIF{$|J.B| = 1$}
        \STATE Reject singleton $J.B[0]$
    \ELSIF{$J.\text{allow\_blame}$}
        \STATE Invoke \textsc{CausalBlameAndSplit}($J, T$) \COMMENT{Alg.~\ref{alg:blame}}
    \ELSE
        \STATE Bisect $J.B$ into pending, prioritizing half with $J.\text{next\_suspect}$
    \ENDIF
\ENDIF
\end{algorithmic}
\end{algorithm}

\subsection{Fail-Safe 1-Step AST/Traceback Suspect Ranking}
When retries are denied or exhausted on a multi-PR batch $B$ with \texttt{allow\_blame=True}, \texttt{ASTStackCausalRanker} (Algorithm~\ref{alg:blame}) ranks suspects statically. From traceback $T$, it parses maximum 1-based frame indices $(d_f, d_s)$ and failing test IDs $\mathcal{T}_F$. For each $t \in \mathcal{T}_F$, 1-hop expansion of bare-name AST calls yields direct callees $D(t)$, direct-callee defining files $F_D(t)$, reachable symbols $S(t)$, and reachable files $F(t)$. Inverse-fanout specificity down-weights broad tests while applying an identical $1.5\times$ direct-callee weight to both file and function terms so the file-level ablation isolates granularity alone: $\sigma_F(f)=\sum_{t:\, f\in F(t)} (1.5 \text{ if } f\in F_D(t) \text{ else } 1.0)/|F(t)|$ and $\sigma_S(s)=\sum_{t:\, s\in S(t)} (1.5 \text{ if } s\in D(t) \text{ else } 1.0)/|S(t)|$.

For each modified file $f \in F(p)$, diffing normalized AST dumps of synchronous \texttt{FunctionDef} nodes keyed by bare name yields changed symbols $\Delta_{\text{AST}}(p, f)$ (comments and formatting leave dumps invariant; docstrings alter dumps; syntax errors emit a sentinel; module-level code, \texttt{async} functions, attribute calls, and dynamic dispatch lie outside this subset). AST-invariant files add $0.01$; functionally modified files add base $1.0$, frame bonus $1.0+0.2\,d_f$ if implicated, file specificity $2.0\,\sigma_F(f)$, and per changed symbol $s$: $4.0+0.5\,d_s$ if implicated plus $10.0\,\sigma_S(s)$ (tie-break: PR ID).

SSMQ enqueues remainder $(B\setminus\{s_1\}, \text{allow\_blame}=\text{False}, \text{hint}=s_2)$ and singleton $\{s_1\}$ to pending (dropping the hint on further bisection). Across $W$ parallel workers, $\{s_1\}$ and $B\setminus\{s_1\}$ may complete in either order; if $B\setminus\{s_1\}$ fails, Algorithm~\ref{alg:flaky} routes it directly to $s_2$-prioritized bisection. Algorithm~\ref{alg:blame} uses executable test output in the benchmark and Git replay. The simulator draws suspect hits from configured probabilities because public merge records lack tracebacks.

\begin{algorithm}[t]
\caption{1-Step AST/Traceback Blame (\textsc{CausalBlameAndSplit})}
\label{alg:blame}
\begin{algorithmic}[1]
\REQUIRE Failing batch $B$ ($|B| \ge 2$), traceback $T$, base repository ASTs
\STATE Parse max 1-based frame indices $(d_f, d_s)$ and failing tests $\mathcal{T}_F$
\STATE Expand 1-hop bare-name calls from $t \in \mathcal{T}_F$ to get $D(t), F_D(t), S(t), F(t)$
\STATE $\sigma_F(f)=\sum_{t:\, f\in F(t)}\frac{1.5^{\mathbb{I}[f\in F_D(t)]}}{|F(t)|}$, $\sigma_S(s)=\sum_{t:\, s\in S(t)}\frac{1.5^{\mathbb{I}[s\in D(t)]}}{|S(t)|}$
\FORALL{$p \in B$}
    \STATE $\text{score}(p) \leftarrow 0$
    \FORALL{$f \in F(p)$}
        \STATE $\Delta \leftarrow \text{ChangedFuncSymbols}(p, f)$
        \IF{$\Delta = \emptyset$}
            \STATE $\text{score}(p) \leftarrow \text{score}(p) + 0.01$
        \ELSE
            \STATE $\text{score}(p) \leftarrow \text{score}(p) + 1.0 + \mathbb{I}[f\in\mathcal{F}_T](1.0 + 0.2\,d_f) + 2.0\,\sigma_F(f)$
            \STATE \qquad $+\sum_{s \in \Delta}\big(\mathbb{I}[s\in\mathcal{S}_T](4.0 + 0.5\,d_s) + 10.0\,\sigma_S(s)\big)$
        \ENDIF
    \ENDFOR
\ENDFOR
\STATE $(s_1, s_2) \leftarrow$ Top-2 PRs in $B$ by descending $\text{score}(p)$ (tie-break: PR ID)
\STATE Enqueue $(B \setminus \{s_1\}, \text{allow\_blame}=\text{False}, \text{hint}=s_2)$ and $(\{s_1\})$ to pending
\end{algorithmic}
\end{algorithm}

\subsection{Scoped Formal Properties}
Three propositions bound SSMQ under exact-tree verification.

\begin{proposition}[Exact-Tree Commit Safety]
\label{prop:safety}
Under an initially clean mainline $M$, deterministic test verification of any tested tree $T$, false-failure-only flakes, and atomic check-and-commit transitions requiring $M_0 \cup J.B = M \cup B_{\text{commit}}$ (Algorithm~\ref{alg:flaky}, line~2), every committed mainline state $M'$ has directly passed the test suite with zero oracle assumptions.
\end{proposition}
\begin{proof}[Proof Sketch]
Every transition $M \leftarrow M \cup B_{\text{commit}}$ requires $R.\text{pass}=\text{True}$ on exact tree $M_0 \cup J.B = M \cup B_{\text{commit}}$. For unstacked batches ($B_{\text{commit}}=J.B$), this reduces to $M = M_0$; for a single-timeline speculative prefix stack, $M_0$ includes the exact upstream prefixes that committed prior to $J$. Induction over commits preserves tested-tree safety without any conflict-oracle check. Conversely, multi-lane or per-PR DAG policies that omit concurrent commits from other lanes from $M_0$ cannot satisfy $M_0 \cup J.B = M \cup B_{\text{commit}}$ after another lane commits without cross-lane invalidation.
\end{proof}

\begin{proposition}[Fresh-Batch Dispatch Bound]
\label{prop:liveness}
If $N$ PRs precede PR $p$ in append-only ready queue $Q$ on arrival and pending retries/splits finish in finite work so fresh batches continue forming, Algorithm~\ref{alg:packing} dispatches $p$ within at most $N+1$ fresh batch formations.
\end{proposition}
\begin{proof}[Proof Sketch]
Each Algorithm~\ref{alg:packing} call pops $Q[0]$ (line~1) and preserves relative order in $Q'$, reducing the PRs ahead of $p$ by at least $1$ per fresh batch until $p=Q[0]$ within $N$ formations and $p$ dispatches next (bounding fresh-batch formations, not wall-clock latency under worker contention).
\end{proof}

\begin{proposition}[Fixed-Base Recovery Depth Bound]
\label{prop:complexity}
For a failing batch $B$ ($K\ge 4$) on a fixed base with deterministic observations, no retries or stale rechecks, balanced splitting, and $c\ge 1$ independent unary culprits (distinct from pairwise interaction edges), let $a=\lceil\log_2(K-1)\rceil$, $b=\lceil\log_2 K\rceil$, and $\alpha_1$ be Algorithm~\ref{alg:blame}'s Top-1 hit rate. When $c=1$ (sole unary culprit $u$), expected dependency depth to clear $B\setminus\{u\}$ satisfies:
\begin{equation}
\mathbb{E}[D_{\text{SSMQ}} \mid c=1] \le U(\alpha_1) \triangleq \alpha_1 \cdot 1 + (1 - \alpha_1)(1 + a).
\label{eq:blame_rounds}
\end{equation}
The upper bound $U(\alpha_1)$ beats balanced bisection's worst-case depth $b$ iff $\alpha_1 > (1+a-b)/a$ (sufficiently $\alpha_1 > 1/b$, for instance, $1/3$ at $K=8$). For $c\ge 2$ independent unary culprits, evicting any single $s_1$ leaves $\ge c-1\ge 1$ unary culprits in $B\setminus\{s_1\}$, forcing bisection fallback with worst-case depth at most $1+a$.
\end{proposition}
\begin{proof}[Proof Sketch]
For $c=1$, Top-1 eviction clears $B\setminus\{u\}$ in depth $1$ with probability $\alpha_1$; on a miss, bisecting $K-1$ PRs adds depth at most $a$ (less when $u$ enters a smaller odd half), giving Eq.~\eqref{eq:blame_rounds}, where $U(\alpha_1)<b \iff \alpha_1>(1+a-b)/a$. For $c\ge 2$ independent unary culprits, $B\setminus\{s_1\}$ always fails (whereas for interaction edges $(s,x),(s,y)$ without unary faults, evicting shared endpoint $s$ clears both). Simulator metric \texttt{mean\_blame\_rounds} also counts depth-0 singletons and retry rounds, and worker contention can serialize siblings.
\end{proof}
